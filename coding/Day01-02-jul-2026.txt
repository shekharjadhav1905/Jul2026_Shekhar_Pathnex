 provider "aws" {   
     region = "us-east-1"
 }

 resource "aws_instance" "PathnexEC2" {
     ami = "ami-0abcd1234abcd1234"
     instance_type = "t2.micro"

     tags = {
         Name = "PathnexEC2"
     }
 }

# provider can be changed Azure gcp
# provider is a fixed word

# region : Can be changed - us-east-1 ap-south-1, eu-west-1

# resource : "TYPE" "Name" 
# TYPE : "aws_instance" "Name": Any Name

# ami(Amazon machine image) : which OS/ Template to use 

# tags : Used as labels to identify resources 

# 1. Use AWS.
# 2. Work in the us-east-1 region.
# 3. Create an EC2 virtual machine.
# 4. Call it PathnexEC2 inside Terraform.
# 5. Use this machine image (AMI).
# 6. Make it a t2.micro server.
# 7. Add a tag called Name.
# 8. Set the Name tag to PathnexEC2.
